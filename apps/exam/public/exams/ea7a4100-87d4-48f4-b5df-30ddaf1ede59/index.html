#[VTDmgSystem.java 虚拟伤害系统主类]
import java.util.HashMap;
import java.util.UUID;
public class VTDmgSystem {
privatestatic HashMap<UUID, Integer> dmgMap= new HashMap<>(); #[存储玩家虚拟伤害值 键为玩家 UUID 值为累积伤害]
privatestatic final int HEART_THRESHOLD= 100; #[每颗心对应的虚拟伤害阈值]
public static void addDamage(Player player, int amount) { #[增加虚拟伤害 并触发检测]
UUID id= player.getUniqueId();
int current = dmgMap.getOrDefault(id, 0);
current += amount;
dmgMap.put(id, current);
if (current >= HEART_THRESHOLD) { #[达到阈值则扣除生命值]
deductHeart(player);
dmgMap.put(id, 0);
}
notifyPlayer(player, current); #[通知玩家]
}
privatestatic void deductHeart(Player player) { #[扣除玩家一颗心]
double maxHealth = player.getAttribute(Attribute.GENERIC_MAX_HEALTH).getValue();
player.setHealth(player.getHealth() - 2.0); #[2.0 对应游戏中 1 颗心]
}
privatestatic void notifyPlayer(Player player, int current) { #[通知玩家当前累积伤害]
player.sendMessage("§c[受击监测] 当前累积伤害:§e" + current + "§c/100");
}
public static int getDamage(Player player){ #[查询玩家当前伤害值]
return dmgMap.getOrDefault(player.getUniqueId(), 0);
}
public static void clearDamage(Player player) { #[清零玩家伤害值]
dmgMap.put(player.getUniqueId(), 0);
player.sendMessage("§a[系统] 累积伤害已清零");
}
}

血时机延后
D. 管理员执行 vtdmg clr 后，JAVA 层 dmgMap 与 scoreboard 层 vt.dmg 均可被清零，但玩家已损失的生命值不会恢复，体现虚拟
伤害与原版生命的分离设计

privateHashMap<String, Integer> activeTime; // key:UID, value:在线时长(秒)
privateRandom random = new Random();
public String assignUid(Stringplayer) {
if (uidMap.containsKey(player)) {
return uidMap.get(player);
}
String newUid = "MC-" + String.format("%06d", random.nextInt(1000000));
uidMap.put(player, newUid);
scoreMap.put(newUid, 0);
return newUid;
}
public void calcScore(Stringuid, String type, int value) {
int baseScore = 0;
switch (type) {
case"active":baseScore = value/ 60; break;
case"mine": baseScore= value * 2; break;
case"kill": baseScore = value * 5; break;
}
scoreMap.put(uid, scoreMap.getOrDefault(uid, 0) + baseScore);
}
public List<String> getRankList(int top) {
return scoreMap.entrySet().stream()
.sorted((e1, e2) ->e2.getValue() - e1.getValue())
.limit(top)
.map(e ->e.getKey() + "|" + e.getValue())
.collect(Collectors.toList());
}
